Use pagination for multi-page lists. Bind `page` on `Root` and render pages from the children snippet
(`pages`, `currentPage`). Prefer `Previous` / `Next` (legacy `PrevButton` / `NextButton` still work).

```svelte
<script lang="ts">
	import * as Pagination from '$lib/components/ui/pagination/index.js';
</script>

<Pagination.Root count={100} perPage={10}>
	{#snippet children({ pages, currentPage })}
		<Pagination.Content>
			<Pagination.Item>
				<Pagination.Previous />
			</Pagination.Item>
			{#each pages as page (page.key)}
				{#if page.type === 'ellipsis'}
					<Pagination.Item>
						<Pagination.Ellipsis />
					</Pagination.Item>
				{:else}
					<Pagination.Item>
						<Pagination.Link {page} isActive={currentPage === page.value}>
							{page.value}
						</Pagination.Link>
					</Pagination.Item>
				{/if}
			{/each}
			<Pagination.Item>
				<Pagination.Next />
			</Pagination.Item>
		</Pagination.Content>
	{/snippet}
</Pagination.Root>
```
