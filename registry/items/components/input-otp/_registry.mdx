Use input-otp for one-time codes. Built on bits-ui PinInput. Compose `Root` with `Group`,
`Slot`, and optional `Separator` inside the cells snippet.

```svelte
<script lang="ts">
	import * as InputOTP from '$lib/components/ui/input-otp/index.js';
</script>

<InputOTP.Root maxlength={6}>
	{#snippet children({ cells })}
		<InputOTP.Group>
			{#each cells.slice(0, 3) as cell (cell)}
				<InputOTP.Slot {cell} />
			{/each}
		</InputOTP.Group>
		<InputOTP.Separator />
		<InputOTP.Group>
			{#each cells.slice(3, 6) as cell (cell)}
				<InputOTP.Slot {cell} />
			{/each}
		</InputOTP.Group>
	{/snippet}
</InputOTP.Root>
```
